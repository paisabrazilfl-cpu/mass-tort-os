LAPLACIAN_KERNEL = np.array([
    [0.05, 0.20, 0.05],
    [0.20,-1.00, 0.20],
    [0.05, 0.20, 0.05]
], dtype=np.float64)

def laplacian(field, bc="periodic"):
    """
    Discrete Laplacian with intrinsic boundary condition.
    bc:
      - 'periodic' : toroidal wrap
      - 'neumann'  : no-flux via symmetric reflection
      - 'dirichlet': fixed-zero exterior
    """
    if bc == "periodic":
        return convolve2d(field, LAPLACIAN_KERNEL, mode="same", boundary="wrap")
    elif bc == "neumann":
        return convolve2d(field, LAPLACIAN_KERNEL, mode="same", boundary="symm")
    elif bc == "dirichlet":
        return convolve2d(field, LAPLACIAN_KERNEL, mode="same", boundary="fill", fillvalue=0.0)
    else:
        raise ValueError(f"Unknown boundary condition: {bc}")

def gray_scott_step(u, v, C, dt=1.0):
    """
    Intrinsic constraint-driven dynamics.
    C defines the lawful transformation regime itself.
    
    C fields:
      Du, Dv : diffusion coefficients
      F, k   : feed and kill parameters
      bc     : boundary condition mode
    """
    Du = C["Du"]
    Dv = C["Dv"]
    F  = C["F"]
    k  = C["k"]
    bc = C["bc"]

    lap_u = laplacian(u, bc)
    lap_v = laplacian(v, bc)

    uvv = u * v * v

    du = Du * lap_u - uvv + F * (1.0 - u)
    dv = Dv * lap_v + uvv - (F + k) * v

    u_next = np.clip(u + dt * du, 0.0, 1.0)
    v_next = np.clip(v + dt * dv, 0.0, 1.0)

    return u_next, v_next

def f_ontonomic(state, C, shape=(64, 64), dt=1.0):
    """
    State transition x_{t+1} = f(x_t; C)
    """
    N = shape[0] * shape[1]
    u = state[:N].reshape(shape)
    v = state[N:].reshape(shape)

    u_next, v_next = gray_scott_step(u, v, C, dt=dt)
    return np.concatenate([u_next.ravel(), v_next.ravel()])

def make_initial_state(shape=(64, 64), noise=0.02):
    """
    Standard Gray-Scott seed:
    - background u ~ 1
    - background v ~ 0
    - central perturbation
    - small stochastic noise for run variability
    """
    h, w = shape
    u = np.ones(shape, dtype=np.float64)
    v = np.zeros(shape, dtype=np.float64)

    cx, cy = h // 2, w // 2
    r = min(h, w) // 10

    u[cx-r:cx+r, cy-r:cy+r] = 0.50
    v[cx-r:cx+r, cy-r:cy+r] = 0.25

    u += noise * np.random.randn(h, w)
    v += noise * np.random.randn(h, w)

    u = np.clip(u, 0.0, 1.0)
    v = np.clip(v, 0.0, 1.0)

    return np.concatenate([u.ravel(), v.ravel()])

def normalize_probability(x):
    p = np.abs(x.astype(np.float64)).ravel()
    s = np.sum(p)
    if s <= 1e-15:
        return np.ones_like(p) / len(p)
    return p / s

def shannon_entropy_prob(p):
    p = p[p > 0]
    return -np.sum(p * np.log2(p))

def spatial_entropy(field, bins=64):
    """
    Histogram-based spatial intensity entropy.
    This is NOT thermodynamic entropy.
    It is a descriptive distributional metric.
    """
    hist, _ = np.histogram(field.ravel(), bins=bins, range=(0.0, 1.0), density=False)
    hist = hist.astype(np.float64)
    hist /= (np.sum(hist) + 1e-15)
    return shannon_entropy_prob(hist)

def radial_power_spectrum_metric(field):
    """
    Measures concentration of spatial frequency power.
    Higher value => more nontrivial spatial structure concentration.
    """
    F = np.fft.fftshift(np.fft.fft2(field))
    P = np.abs(F) ** 2

    h, w = field.shape
    cy, cx = h // 2, w // 2
    y, x = np.indices((h, w))
    r = np.sqrt((x - cx) ** 2 + (y - cy) ** 2).astype(np.int32)

    max_r = min(cx, cy)
    radial = np.zeros(max_r + 1, dtype=np.float64)
    counts = np.zeros(max_r + 1, dtype=np.float64)

    for i in range(h):
        for j in range(w):
            rr = r[i, j]
            if rr <= max_r:
                radial[rr] += P[i, j]
                counts[rr] += 1

    radial /= (counts + 1e-15)
    radial /= (np.sum(radial) + 1e-15)

    # spectral concentration away from DC only
    if len(radial) <= 2:
        return 0.0
    nonzero = radial[1:]
    return np.sum(nonzero ** 2)

def pattern_energy(field):
    """
    Mean squared gradient as a simple structure-energy proxy.
    """
    gx = np.diff(field, axis=1, append=field[:, -1:])
    gy = np.diff(field, axis=0, append=field[-1:, :])
    return np.mean(gx**2 + gy**2)

def convergence_rate(traj):
    """
    Mean terminal state change magnitude.
    Lower => more stabilized late dynamics.
    """
    diffs = []
    for i in range(len(traj) - 1):
        diffs.append(np.linalg.norm(traj[i+1] - traj[i]))
    diffs = np.array(diffs, dtype=np.float64)
    tail = diffs[max(0, len(diffs) - 20):]
    return np.mean(tail) if len(tail) > 0 else np.nan

def kl_divergence_from_states(x, y):
    """
    KL divergence over normalized absolute state amplitudes.
    Descriptive divergence only.
    """
    p = normalize_probability(x) + 1e-15
    q = normalize_probability(y) + 1e-15
    return np.sum(p * np.log(p / q))

def extract_metrics(final_state, reference_state, traj, shape=(64, 64)):

        "u_pattern_energy": pattern_energy(u),
        "v_pattern_energy": pattern_energy(v),
        "terminal_convergence": convergence_rate(traj),
        "kl_vs_reference": kl_divergence_from_states(final_state, reference_state),
    }

def simulate(C, shape=(64, 64), steps=300, dt=1.0, noise=0.02):
    state = make_initial_state(shape=shape, noise=noise)
    traj = [state.copy()]

    for _ in range(steps):
        state = f_ontonomic(state, C, shape=shape, dt=dt)
        traj.append(state.copy())

    return np.array(traj)

def run_experiment(regimes, n_runs=40, shape=(64, 64), steps=300, dt=1.0, noise=0.02):
    """
    For each run:
      - create shared initial state seed path via global RNG
      - simulate each regime independently
      - compare against reference regime (periodic)
    """
    results = {reg["name"]: [] for reg in regimes}

    for _ in range(n_runs):
        # Shared initial condition across regimes for paired comparison
        base_state = make_initial_state(shape=shape, noise=noise)

        trajectories = {}
        for reg in regimes:
            state = base_state.copy()
            traj = [state.copy()]
            for _ in range(steps):
                state = f_ontonomic(state, reg, shape=shape, dt=dt)
                traj.append(state.copy())
            trajectories[reg["name"]] = np.array(traj)

        reference_final = trajectories["periodic"][-1]

        for reg in regimes:
            name = reg["name"]
            traj = trajectories[name]
            final_state = traj[-1]
            metrics = extract_metrics(final_state, reference_final, traj, shape=shape)
            results[name].append(metrics)

    return results

def cohens_d_paired(x, y):
    d = np.array(x) - np.array(y)
    return np.mean(d) / (np.std(d, ddof=1) + 1e-15)

def bonferroni_correction(pvals):
    pvals = np.array(pvals, dtype=np.float64)
    m = len(pvals)
    return np.minimum(pvals * m, 1.0)

def compare_regimes(results, baseline="periodic", target="neumann"):
    """
    Paired statistical comparison because runs share the same initial condition.
    """
    metric_names = list(results[baseline][0].keys())

    out = []
    raw_p = []

    for metric in metric_names:
        x = np.array([r[metric] for r in results[target]], dtype=np.float64)
        y = np.array([r[metric] for r in results[baseline]], dtype=np.float64)

        stat, p = stats.ttest_rel(x, y, nan_policy="omit")
        d = cohens_d_paired(x, y)

        raw_p.append(p)
        out.append({
            "metric": metric,
            "target_mean": float(np.mean(x)),
            "baseline_mean": float(np.mean(y)),
            "delta": float(np.mean(x - y)),
            "t_stat": float(stat),
            "p_raw": float(p),
            "effect_size_dz": float(d),
        })

    corrected = bonferroni_correction(raw_p)
    for i in range(len(out)):
        out[i]["p_bonf"] = float(corrected[i])

    return out

def summarize_comparison(comparison, alpha=0.05):
    lines = []
    for row in comparison:
        sig = "YES" if row["p_bonf"] < alpha else "NO"
        lines.append(
            f"{row['metric']}: "
            f"Δ={row['delta']:.6f}, "
            f"p_raw={row['p_raw']:.6g}, "
            f"p_bonf={row['p_bonf']:.6g}, "
            f"dz={row['effect_size_dz']:.3f}, "
            f"significant={sig}"
        )
    return "\n".join(lines)
